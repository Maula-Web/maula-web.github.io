function doPost(e) {
  try {
    var contents = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    var data = JSON.parse(contents);
    
    // CASO 1: CONSULTAR RESPUESTA DE LA ADMINISTRACIÓN EN GMAIL
    // Si la acción es check_reply O no viene fileContent (evita cualquier fallo si no es envío)
    if (data.action === 'check_reply' || !data.fileContent) {
      return procesarConsultaRespuesta(data.ticketId, data.jornada, data.isTest || data.esPrueba || false);
    }
    
    // CASO 2: ENVIAR FICHERO .TXT DE 35 APUESTAS POR CORREO REAL
    var fileName = data.fileName || 'apuestas.txt';
    var blob = Utilities.newBlob(data.fileContent, 'text/plain', fileName);
    GmailApp.sendEmail(data.to, data.subject, data.body, {
      name: 'Peña Los Maulas',
      attachments: [blob]
    });
    
    return ContentService.createTextOutput(JSON.stringify({ 
      ok: true, 
      sentAt: new Date().toISOString(),
      to: data.to,
      fileName: fileName,
      ticketId: data.ticketId || null
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      ok: false, 
      error: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var action = e && e.parameter && e.parameter.action;
    var ticketId = e && e.parameter && e.parameter.ticketId;
    var jornada = e && e.parameter && e.parameter.jornada;
    var isTest = e && e.parameter && (e.parameter.isTest === 'true' || e.parameter.esPrueba === 'true');
    
    if (action === 'check_reply' || ticketId || jornada) {

}

/**
 * Función auxiliar para obtener o crear una carpeta dentro de otra (o en la raíz de Drive)
 */
function obtenerOCrearCarpeta(nombre, parentFolder) {
  var folders = parentFolder ? parentFolder.getFoldersByName(nombre) : DriveApp.getFoldersByName(nombre);
  if (folders.hasNext()) {
    return folders.next();
  }
  return parentFolder ? parentFolder.createFolder(nombre) : DriveApp.createFolder(nombre);
}

/**
 * Deduce la temporada a partir del ticketId (ej. MAULAS-2627-... -> TEMPORADA 2026-2027)
 */
function deducirTemporada(ticketId) {
  if (ticketId) {
    var m = ticketId.match(/MAULAS-(\d{2})(\d{2})-/i);
    if (m) {
      return 'TEMPORADA 20' + m[1] + '-20' + m[2];
    }
  }
  return 'TEMPORADA 2026-2027';
}

/**
 * Busca en la bandeja de Penalosmaulas@gmail.com el hilo de sellado
 * y extrae la respuesta del lotero (texto y archivo resguardo adjunto).
 * Guarda el resguardo en Google Drive en:
 * RESGUARDOS QUINIELAS MAULAS / TEMPORADA 2026-2027 / [RESGUARDOS DE PRUEBA]
 */
function procesarConsultaRespuesta(ticketId, jornada, isTest) {
  try {
    var threads = [];

    // 1. Búsqueda por ticketId exacto si viene indicado
    if (ticketId) {
      threads = GmailApp.search('"' + ticketId + '"', 0, 5);
      if (!threads || threads.length === 0) {
        threads = GmailApp.search('subject:"' + ticketId + '"', 0, 5);
      }
      if (!threads || threads.length === 0) {
        var cleanId = ticketId.replace(/[\[\]]/g, '');
        threads = GmailApp.search(cleanId, 0, 5);
      }
    }

    // 2. Si no se encontró por ID exacto, buscar por el número de Jornada (J12 o Jornada 12)
    if (!threads || threads.length === 0) {
      var jNum = jornada;
      if (!jNum && ticketId) {
        var mJ = ticketId.match(/-J(\d+)/i);
        if (mJ) jNum = mJ[1];
      }
      if (jNum) {
        var jPad = ('0' + jNum).slice(-2);
        var qJornada = 'subject:"[MAULAS-" (subject:"-J' + jPad + '" OR subject:"Jornada ' + parseInt(jNum, 10) + '") newer_than:7d';
        threads = GmailApp.search(qJornada, 0, 5);
      }
    }

    // 3. Fallback: El último correo de sellado de Los Maulas de los últimos 7 días
    if (!threads || threads.length === 0) {
      threads = GmailApp.search('subject:"[MAULAS-" newer_than:7d', 0, 5);
    }
    if (!threads || threads.length === 0) {
      threads = GmailApp.search('subject:MAULAS newer_than:7d', 0, 5);
    }

    if (!threads || threads.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        hasReply: false,
        foundThread: false,
        ticketId: ticketId,
        message: 'No se encontró ningún correo con asunto [MAULAS- en Gmail en los últimos 7 días.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var thread = threads[0];
    var messages = thread.getMessages();
    var firstSubject = thread.getFirstMessageSubject();

    // Intentar extraer el ticketId real del asunto del hilo encontrado (ej: [MAULAS-2627-20261009_1913-J12])
    var foundTicketId = ticketId;
    var matchRealId = firstSubject.match(/\[(MAULAS-[^\]]+)\]/);
    if (matchRealId) {
      foundTicketId = matchRealId[1];
    }

    // Comprobar si hay respuesta en el hilo
    // Buscamos hacia atrás el mensaje más reciente que NO sea de penalosmaulas@gmail.com
    var replyMsg = null;
    for (var m = messages.length - 1; m >= 0; m--) {
      var sender = messages[m].getFrom().toLowerCase();
      if (sender.indexOf('penalosmaulas@gmail.com') === -1) {
        replyMsg = messages[m];
        break;
      }
    }

    // Si todos los mensajes son de penalosmaulas (pruebas entre misma cuenta) y hay más de 1 mensaje:
    if (!replyMsg && messages.length > 1) {
      replyMsg = messages[messages.length - 1];
    }

    // Si es un mensaje único pero su asunto es una respuesta (Re: / RV:)
    if (!replyMsg && messages.length === 1 && /^(re:|rv:)/i.test(messages[0].getSubject())) {
      replyMsg = messages[0];
    }

    if (!replyMsg) {
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        hasReply: false,
        foundThread: true,
        ticketId: foundTicketId,
        subject: firstSubject,
        messagesCount: messages.length,
        sentDate: messages[0].getDate().toISOString(),
        message: 'El correo de sellado fue enviado, pero la administración aún no ha respondido.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Procesar adjuntos de forma segura y guardarlos en la jerarquía inteligente de Google Drive
    var attachments = replyMsg.getAttachments();
    var attList = [];
    var folderDestino = null;

    try {
      // 1. Carpeta Raíz: RESGUARDOS QUINIELAS MAULAS
      var folderRaiz = obtenerOCrearCarpeta("RESGUARDOS QUINIELAS MAULAS", null);
      
      // 2. Subcarpeta de Temporada inteligente (ej: TEMPORADA 2026-2027)
      var nombreTemporada = deducirTemporada(foundTicketId);
      var folderTemporada = obtenerOCrearCarpeta(nombreTemporada, folderRaiz);
      
      // 3. Si estamos en pruebas, guardamos dentro de 'RESGUARDOS DE PRUEBA'
      if (isTest) {
        folderDestino = obtenerOCrearCarpeta("RESGUARDOS DE PRUEBA", folderTemporada);
      } else {
        folderDestino = folderTemporada;
      }
    } catch (fErr) {
      folderDestino = null;
    }
    
    for (var i = 0; i < attachments.length; i++) {
      try {
        var att = attachments[i];
        var cType = att.getContentType() || 'application/octet-stream';
        var bytes = att.getBytes();
        var size = bytes ? bytes.length : 0;
        var dataUri = null;
        var driveFileUrl = null;
        var driveFileId = null;
        var originalName = att.getName() || ('adjunto_' + (i + 1));
        var ext = originalName.lastIndexOf('.') !== -1 ? originalName.substring(originalName.lastIndexOf('.')) : '';
        
        // Guardar copia en Google Drive con nombre estandarizado (si hay carpeta disponible)
        if (folderDestino) {
          try {
            var driveFileName = 'RESGUARDO_' + foundTicketId + (attachments.length > 1 ? ('_part' + (i + 1)) : '') + ext;
            var existingFiles = folderDestino.getFilesByName(driveFileName);
            var driveFile = null;
            if (existingFiles.hasNext()) {
              driveFile = existingFiles.next();
            } else {
              driveFile = folderDestino.createFile(att.copyBlob().setName(driveFileName));
              driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            }
            driveFileId = driveFile.getId();
            driveFileUrl = 'https://drive.google.com/uc?export=view&id=' + driveFileId;
          } catch (dErr) {
            // Continúa con fallback de dataUri
          }
        }
        
        // Si pesa menos de 4 MB generamos data URI para preview visual inmediato
        if (bytes && size < 4 * 1024 * 1024) {
          dataUri = 'data:' + cType + ';base64,' + Utilities.base64Encode(bytes);
        }
        
        attList.push({
          name: originalName,
          contentType: cType,
          size: size,
          dataUri: dataUri,
          driveUrl: driveFileUrl,
          driveId: driveFileId
        });
      } catch(attErr) {
        attList.push({
          name: 'Adjunto ' + (i + 1),
          contentType: 'application/octet-stream',
          size: 0,
          dataUri: null,
          driveUrl: null,
          driveId: null
        });
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      hasReply: true,
      foundThread: true,
      ticketId: foundTicketId,
      subject: replyMsg.getSubject(),
      from: replyMsg.getFrom(),
      date: replyMsg.getDate().toISOString(),
      body: replyMsg.getPlainBody().substring(0, 1500),
      attachmentsCount: attachments.length,
      attachments: attList
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      ok: false,
      error: 'Error consultando respuesta: ' + err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
